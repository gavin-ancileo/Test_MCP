🔍 Jira Project Debug - Production

📋 Instructions

1. Open Browser Console:

  • Press F12 to open DevTools
  • Go to Console tab

2. Copy and paste this code into Console:

(async () => {
  // Get your current user info
  const user = JSON.parse(localStorage.getItem('user') || '{}');
  console.log('👤 Current user:', user);

  // Get Cognito ID token for authentication
  const idToken = localStorage.getItem('idToken');
  if (!idToken) {
    console.error('❌ Not logged in! Please login first at https://internal.assistant.leacare.ai');
    return;
  }

  console.log('✅ Authenticated');

  // Call backend MCP tools
  const baseUrl = 'https://internal.assistant.leacare.ai';

  try {
    // Method 1: Get ALL projects (OLD - will show unexpected projects)
    console.log('\\n📊 Method 1: Getting ALL projects (OLD method)...');
    const oldResponse = await fetch(`${baseUrl}/mcp-server/tools/jira_list_projects`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${idToken}`
      },
      body: JSON.stringify({ user_id: user.email })
    });

    const oldData = await oldResponse.json();
    console.log('📋 Old method result:', oldData);

    const oldProjects = oldData.projects || [];
    console.log(`   Total projects (OLD): ${oldProjects.length}`);

    // Method 2: Get only YOUR active projects (NEW - our fix)
    console.log('\\n📊 Method 2: Getting YOUR active projects (NEW method)...');
    const newResponse = await fetch(`${baseUrl}/mcp-server/tools/jira_list_my_active_projects`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${idToken}`
      },
      body: JSON.stringify({ user_id: user.email })
    });

    const newData = await newResponse.json();
    console.log('📋 New method result:', newData);

    const newProjects = newData.projects || [];
    console.log(`   Total projects (NEW): ${newProjects.length}`);

    // Compare
    const newKeys = new Set(newProjects.map(p => p.key));
    const unexpectedProjects = oldProjects.filter(p => !newKeys.has(p.key));

    console.log('\\n' + '='.repeat(80));
    console.log('📊 COMPARISON RESULTS');
    console.log('='.repeat(80));
    console.log(`Old method (ALL projects): ${oldProjects.length}`);
    console.log(`New method (YOUR projects): ${newProjects.length}`);
    console.log(`Unexpected projects: ${unexpectedProjects.length}`);

    if (unexpectedProjects.length > 0) {
      console.log('\\n⚠️  UNEXPECTED PROJECTS (You have NO issues in these):');
      console.log('='.repeat(80));
      unexpectedProjects.forEach(p => {
        console.log(`❌ ${p.key} - ${p.name}`);
      });

      console.log('\\n💡 WHY THIS HAPPENS:');
      console.log('   1. You have "Browse Projects" permission in Jira workspace');
      console.log('   2. Workspace admin gave you access to see ALL projects');
      console.log('   3. But you don\\'t actually work on these projects');

      console.log('\\n✅ SOLUTION:');
      console.log('   The NEW function filters these out automatically!');
      console.log('   Use jira_list_my_active_projects instead of jira_list_projects');
    } else {
      console.log('\\n✅ Perfect! No unexpected projects found.');
    }

    console.log('\\n📋 YOUR ACTIVE PROJECTS:');
    console.log('='.repeat(80));
    newProjects.forEach(p => {
      console.log(`✅ ${p.key} - ${p.name}`);
    });

    // Return results for HTML display
    return {
      oldProjects,
      newProjects,
      unexpectedProjects,
      user
    };

  } catch (error) {
    console.error('❌ Error:', error);
    return null;
  }
})();

3. Press Enter and check the Console output
